## Бэктест (функция)
```python
def sim(path, delay=30, tp1=0, part=0, tp=200, sl=50, hold=300):
    e = next(((t, p) for t, p in path if t >= delay and p > -90), None)
    if e is None or e[1] < -10: return None
    base = 1 + e[1] / 100
    q = [(t - e[0], ((1 + p / 100) / base - 1) * 100) for t, p in path if t >= e[0]]
    first = None
    for i, (t, pct) in enumerate(q):
        nxt = q[min(i + 1, len(q) - 1)][1]
        if first is None and part and tp1 <= pct < tp: first = nxt
        if pct >= tp or pct <= -sl or t >= hold:
            return nxt if first is None else part * first + (1 - part) * nxt
    last = q[-1][1]
    return last if first is None else part * first + (1 - part) * last
```
